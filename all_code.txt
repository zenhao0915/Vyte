// orbyteflow/src/components/SplashScreen.tsx
import React, { useEffect, useState, useRef } from 'react';
import startAnimData from './start_anim.json';

export interface SplashScreenProps {
    /** Callback fired when the splash animation finishes and exits */
    onComplete?: () => void;
    /** Optional override for the JSON animation config */
    animationData?: typeof startAnimData;
    /** Show interactive timeline scrubber at the bottom for debugging */
    showDebugControls?: boolean;
    /** Automatically fade out when complete (default: true) */
    autoDismiss?: boolean;
}

interface InventoryBlock {
    id: string;
    shelf: number;
    x: number;
    y: number;
    w: number;
    h: number;
    color: string;
}

export const SplashScreen: React.FC<SplashScreenProps> = ({
                                                              onComplete,
                                                              animationData = startAnimData,
                                                              showDebugControls = false,
                                                              autoDismiss = true,
                                                          }) => {
    const [elapsedMs, setElapsedMs] = useState<number>(0);
    const [isPlaying, setIsPlaying] = useState<boolean>(true);
    const [isExiting, setIsExiting] = useState<boolean>(false);
    const rafRef = useRef<number | null>(null);
    const lastTimeRef = useRef<number | null>(null);

    const totalDuration = 2400;
    const { colors } = animationData.designTokens;
    const { glyphs, subtitle } = animationData.logoGeometry;

    // Solid Drop Box Palette
    const boxColors = {
        exteriorDark: '#B87B42',
        exteriorLight: '#D49658',
        sealTape: colors.vyteGreen,
        labelPlate: '#FFFFFF',
        statusDot: colors.vyteGreen,
    };

    // --- Key Choreography Milestones (ms) ---
    const T_MORPH_START = 450;
    const T_DROP_START = 850;
    const T_CONTACT = 1150;      // Moment of impact: Box appears & springs!
    const T_TAPE = 1380;
    const T_SHIP_START = 1500;
    const T_LOGO_START = 1680;   // Logo starts sweeping from LEFT

    // 1. Database -> Drawer Morph Progress (0 -> 1)
    const morphProgress = Math.min(1, Math.max(0, (elapsedMs - T_MORPH_START) / 380));

    // 2. Drawer Gravity Drop Progress (0 -> 1)
    const dropLinear = Math.min(1, Math.max(0, (elapsedMs - T_DROP_START) / (T_CONTACT - T_DROP_START)));
    // Gravitational acceleration curve
    const drawerFallY = Math.pow(dropLinear, 2) * 148;

    // 3. Impact & Spring Physics (Active from T_CONTACT onwards)
    const isContacted = elapsedMs >= T_CONTACT;
    const springProgress = Math.min(1, Math.max(0, (elapsedMs - T_CONTACT) / 340));

    // Damped harmonic oscillator math for realistic squash-and-rebound:
    const springDecay = Math.exp(-6 * springProgress);
    const springOsc = Math.sin(3.5 * Math.PI * springProgress);
    const boxSpringY = isContacted ? 16 * springDecay * springOsc : 0;
    const boxScaleY = isContacted ? 1 - 0.16 * springDecay * springOsc : 1;
    const boxScaleX = isContacted ? 1 + 0.09 * springDecay * springOsc : 1;

    // 4. Sealing Tape Pop
    const tapeProgress = Math.min(1, Math.max(0, (elapsedMs - T_TAPE) / 100));

    // 5. Box Rocket Launch to the RIGHT
    const shipProgress = Math.min(1, Math.max(0, (elapsedMs - T_SHIP_START) / 340));
    const boxShipX = Math.pow(shipProgress, 2.6) * 1150;

    // 6. Smooth Logo Glide from the LEFT (Critically Damped Harmonic Arrival)
    const logoProgress = Math.min(1, Math.max(0, (elapsedMs - T_LOGO_START) / 680));
    // Analytical spring formula starting with velocity 0 and settling with a subtle 4% organic cushion
    const logoSpringFactor =
        logoProgress > 0
            ? 1 - Math.exp(-5.2 * logoProgress) * (Math.cos(5.4 * logoProgress) + 0.9 * Math.sin(5.4 * logoProgress))
            : 0;
    const logoArrivalX = -640 * (1 - logoSpringFactor);
    const logoOpacity = Math.min(1, logoProgress * 3.5);

    useEffect(() => {
        if (!isPlaying) {
            lastTimeRef.current = null;
            return;
        }

        const tick = (timestamp: number) => {
            if (lastTimeRef.current === null) {
                lastTimeRef.current = timestamp;
            }
            const delta = timestamp - lastTimeRef.current;
            lastTimeRef.current = timestamp;

            setElapsedMs((prev) => {
                const next = prev + delta;
                if (next >= totalDuration) {
                    setIsPlaying(false);
                    if (autoDismiss && !showDebugControls) {
                        setTimeout(() => setIsExiting(true), 250);
                        setTimeout(() => onComplete?.(), 700);
                    }
                    return totalDuration;
                }
                return next;
            });

            rafRef.current = requestAnimationFrame(tick);
        };

        rafRef.current = requestAnimationFrame(tick);
        return () => {
            if (rafRef.current) cancelAnimationFrame(rafRef.current);
        };
    }, [isPlaying, totalDuration, autoDismiss, showDebugControls, onComplete]);

    const handleReplay = () => {
        setIsExiting(false);
        setElapsedMs(0);
        lastTimeRef.current = null;
        setIsPlaying(true);
    };

    // Fixed, non-shaking inventory SKU blocks
    const inventoryBlocks: InventoryBlock[] = [
        { id: 'b0', shelf: 0, x: 318, y: 132, w: 46, h: 22, color: colors.vyteGreen },
        { id: 'b1', shelf: 0, x: 372, y: 132, w: 48, h: 22, color: colors.inkBlack },
        { id: 'b2', shelf: 0, x: 428, y: 132, w: 54, h: 22, color: colors.vyteGreen },
        { id: 'b3', shelf: 1, x: 318, y: 196, w: 56, h: 22, color: colors.inkBlack },
        { id: 'b4', shelf: 1, x: 382, y: 196, w: 44, h: 22, color: colors.vyteGreen },
        { id: 'b5', shelf: 1, x: 434, y: 196, w: 48, h: 22, color: colors.inkBlack },
    ];

    return (
        <div
            style={{
                position: 'fixed',
                inset: 0,
                zIndex: 9999,
                display: 'flex',
                flexDirection: 'column',
                alignItems: 'center',
                justifyContent: 'center',
                backgroundColor: colors.surfaceOffWhite,
                opacity: isExiting ? 0 : 1,
                pointerEvents: isExiting ? 'none' : 'auto',
                transition: 'opacity 450ms cubic-bezier(0.4, 0, 0.2, 1)',
                fontFamily:
                    '"Plus Jakarta Sans", "Inter", -apple-system, BlinkMacSystemFont, sans-serif',
                userSelect: 'none',
                overflow: 'hidden',
            }}
        >
            {/* Background Ambient Glow */}
            <div
                style={{
                    position: 'absolute',
                    width: '560px',
                    height: '560px',
                    borderRadius: '50%',
                    background: `radial-gradient(circle, ${colors.vyteGreenGlow} 0%, rgba(246,247,244,0) 70%)`,
                    transform: `scale(${isContacted ? 1.25 : 0.85})`,
                    opacity: elapsedMs >= T_LOGO_START ? 0.95 : 0.4,
                    transition: 'transform 500ms ease, opacity 500ms ease',
                    pointerEvents: 'none',
                }}
            />

            {/* Main SVG Vector Canvas */}
            <svg
                viewBox={animationData.logoGeometry.viewBox}
                style={{
                    width: 'min(90vw, 760px)',
                    height: 'auto',
                    overflow: 'visible',
                }}
            >
                <defs>
                    <filter id="soft-depth-shadow" x="-20%" y="-20%" width="140%" height="140%">
                        <feDropShadow dx="0" dy="12" stdDeviation="15" floodColor="#0B0C0E" floodOpacity="0.08" />
                    </filter>
                </defs>

                {/* ============================================================
            GROUP 1: DRAWER & DROP BOX (Ships away to the RIGHT)
           ============================================================ */}
                {elapsedMs < T_LOGO_START + 250 && (
                    <g
                        style={{
                            transform: `translate(${boxShipX}px, 0px)`,
                            opacity: shipProgress > 0.85 ? 1 - (shipProgress - 0.85) * 6.6 : 1,
                        }}
                    >
                        {/* DRAWER UNIT */}
                        <g
                            filter="url(#soft-depth-shadow)"
                            style={{
                                transformOrigin: '400px 180px',
                                transform: `translate(0px, ${drawerFallY + boxSpringY}px) scaleY(${boxScaleY}) scaleX(${boxScaleX})`,
                            }}
                        >
                            {/* Outer Cabinet Frame */}
                            <g style={{ opacity: morphProgress }}>
                                <rect x={286} y={108} width={228} height={158} rx={22} fill={colors.inkBlack} />
                                <rect x={294} y={116} width={212} height={142} rx={16} fill="#FFFFFF" />
                            </g>

                            {/* 2 Database / Drawer Shelves */}
                            {[0, 1].map((idx) => {
                                const tierY = 126 + idx * 64;
                                const dbEnterProgress = Math.min(1, Math.max(0, (elapsedMs - idx * 80) / 280));

                                return (
                                    <g
                                        key={`tier-${idx}`}
                                        style={{
                                            transform: `translateY(${(1 - dbEnterProgress) * -22}px)`,
                                            opacity: dbEnterProgress,
                                        }}
                                    >
                                        {/* Shelf tray base */}
                                        <rect
                                            x={304}
                                            y={tierY}
                                            width={192}
                                            height={48}
                                            rx={morphProgress > 0.5 ? 10 : 24}
                                            fill={morphProgress > 0.5 ? colors.drawerTrack : colors.inkBlack}
                                        />

                                        {/* Top elliptical platter (fades out as drawer appears) */}
                                        <ellipse
                                            cx={400}
                                            cy={tierY + 12}
                                            rx={80}
                                            ry={10}
                                            fill="#1E222A"
                                            style={{ opacity: 1 - morphProgress }}

    async createProduct(payload: NewProductForm) {
        const res = await fetch(`${API_BASE_URL}/products`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload),
        });
        return res.json();
    },

    // Module 1: Approve Restock & Contact Supplier
    async approveRestock(productId: string, approvedByRole: UserRole, orderQty: number) {
        const res = await fetch(`${API_BASE_URL}/restock/approve`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ productId, approvedByRole, finalOrderQty: orderQty }),
        });
        return res.json();
    },

    // Module 4: Approve Quality / Expiry AI Solution
    async resolveQualityAlert(qcId: string, approvedByRole: UserRole, actionLabel: string) {
        const res = await fetch(`${API_BASE_URL}/qc/resolve`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ qcId, approvedByRole, selectedAction: actionLabel }),
        });
        return res.json();
    },

    // Module 5: Verify AI Invoice Categorization

        const res = await fetch(`${API_BASE_URL}/invoices/verify`, {
            method: 'POST',

    const isLow = Number(newProd.stock) <= Number(newProd.minThreshold);
    const created: Product = {
      id: `P-${Date.now()}`,
      sku: newProd.sku || `SKU-${Math.floor(1000 + Math.random() * 9000)}`,
      name: newProd.name,
      category: newProd.category,
      industry: newProd.industry,
      uom: newProd.uom,
      stock: Number(newProd.stock),
      minThreshold: Number(newProd.minThreshold),
      reorderQty: Number(newProd.reorderQty),
      priceRM: Number(newProd.priceRM),
      supplierName: newProd.supplierName || 'Preferred Orbyte Supplier',
      supplierContact: newProd.supplierContact || '+6012-9998888',
      performance: 'Good',
      visibility: true,
      batchNo: newProd.industry === 'FNB_PERISHABLE' ? newProd.extraAttr : undefined,
      variantSpec: newProd.industry !== 'FNB_PERISHABLE' ? newProd.extraAttr : undefined,
      restockStatus: isLow ? 'TRIGGERED_PENDING_APPROVAL' : 'HEALTHY',
    };
    setProducts([created, ...products]);
    addAuditLog(
        'Module 1 (Inventory)',
        `Added new ${created.industry} product "${created.name}". ${isLow ? 'Low-stock alert auto-triggered!' : ''}`
    );
    setShowAddModal(false);
  };

  // Module 4: Approve Quality / Expiry AI Recommendation
  const handleApproveQC = (qcId: string) => {
    if (role !== 'SME_OWNER') {
      alert('Orbyte RBAC Restriction: Only SME_OWNER can approve QC write-offs or promotional markdowns.');
      return;
    }
    setQcAlerts((prev) =>
        prev.map((q) => {
          if (q.id === qcId) {
            addAuditLog('Module 4 (Quality AI)', `Approved AI solution for ${q.productName} (${q.actionLabel}).`);
            return { ...q, status: 'APPROVED_RESOLVED' };
          }
          return q;
        })
    );
  };

  // Module 5: Verify AI Categorized Invoice
  const handleVerifyInvoice = (invId: string) => {
    setInvoices((prev) =>
        prev.map((inv) => {
          if (inv.id === invId) {
            addAuditLog('Module 5 (Invoice AI)', `Verified AI category "${inv.aiCategory}" for ${inv.invNo}.`);
            return { ...inv, status: 'VERIFIED' };
          }
          return inv;
        })
    );
  };

  // Module 5: Simulate Upload & AI Categorize Invoice
  const handleSimulateNewInvoice = () => {
    const newInv: InvoiceItem = {
      id: `INV-${Date.now()}`,
      invNo: `SUP-2026-${Math.floor(1000 + Math.random() * 9000)}`,
      type: 'SUPPLIER_PAYABLE',
      party: 'EcoPack Malaysia Sdn Bhd',
      date: '20 Oct 2026',
      amountRM: 1350.0,
      aiCategory: 'Packaging & Sustainable Supplies',
      aiConfidence: 96,
      paid: false,
      status: 'NEEDS_REVIEW',
    };
    setInvoices([newInv, ...invoices]);
    addAuditLog('Module 5 (Invoice AI)', `AI parsed & categorized uploaded invoice ${newInv.invNo} (96% confidence).`);
  };

  const filteredProducts = products.filter((p) => {
    const matchesInd = industryFilter === 'ALL' || p.industry === industryFilter;
    const matchesSearch =
        p.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
        p.sku.toLowerCase().includes(searchQuery.toLowerCase());
    return matchesInd && matchesSearch;
  });

  return (
      <>
        {!splashDone && (
            <SplashScreen
                onComplete={() => setSplashDone(true)}
                showDebugControls={false} // Set to true to scrub through Figma keyframes!
            />
        )}

        <div className="min-h-screen bg-[#F5F7FA] flex font-sans text-slate-800">
          <Sidebar
              activeTab={activeTab}
              setActiveTab={setActiveTab}
              industryFilter={industryFilter}
              setIndustryFilter={setIndustryFilter}
              searchQuery={searchQuery}
              setSearchQuery={setSearchQuery}
              pendingRestockCount={products.filter((p) => p.restockStatus === 'TRIGGERED_PENDING_APPROVAL').length}
              pendingQcCount={qcAlerts.filter((q) => q.status === 'PENDING_APPROVAL').length}
              pendingInvoiceCount={invoices.filter((i) => i.status === 'NEEDS_REVIEW').length}
              totalProductCount={products.length}
          />

          <main className="flex-1 flex flex-col overflow-y-auto">
            <HeaderBar activeTab={activeTab} role={role} setRole={setRole} />

            <div className="p-8 space-y-6">
              <KpiCards
                  totalSkus={products.length}
                  pendingRestocks={products.filter((p) => p.restockStatus === 'TRIGGERED_PENDING_APPROVAL').length}
                  pendingQcBatches={qcAlerts.filter((q) => q.status === 'PENDING_APPROVAL').length}
                  totalInvoices={invoices.length}
              />

              {activeTab === 'mod1' && (
                  <RestockModule
                      products={filteredProducts}
                      onApproveRestock={handleApproveRestock}
                      onOpenAddModal={() => setShowAddModal(true)}
                  />
              )}

              {activeTab === 'mod4' && (
                  <QualityControlModule qcAlerts={qcAlerts} onApproveQC={handleApproveQC} />
              )}

              {activeTab === 'mod5' && (
                  <InvoiceModule
                      invoices={invoices}
                      onVerifyInvoice={handleVerifyInvoice}
                      onSimulateUploadInvoice={handleSimulateNewInvoice}
                  />
              )}

              {activeTab === 'mod2_3' && <SalesAndOrderPlaceholder />}

              <AuditLogPanel logs={auditLogs} role={role} />
            </div>
          </main>

          <AddProductModal
              isOpen={showAddModal}
              onClose={() => setShowAddModal(false)}
              onSaveProduct={handleSaveProduct}
          />
        </div>
      </>
  );
}

# React + TypeScript + Vite

This template provides a minimal setup to get React working in Vite with HMR and some Oxlint rules.

Currently, two official plugins are available:

- [@vitejs/plugin-react](https://github.com/vitejs/vite-plugin-react/blob/main/packages/plugin-react) uses [Oxc](https://oxc.rs)
- [@vitejs/plugin-react-swc](https://github.com/vitejs/vite-plugin-react/blob/main/packages/plugin-react-swc) uses [SWC](https://swc.rs/)

## React Compiler

The React Compiler is not enabled on this template because of its impact on dev & build performances. To add it, see [this documentation](https://react.dev/learn/react-compiler/installation).

## Expanding the Oxlint configuration

If you are developing a production application, we recommend enabling type-aware lint rules by installing `oxlint-tsgolint` and editing `.oxlintrc.json`:

```json
{
  "$schema": "./node_modules/oxlint/configuration_schema.json",
  "plugins": ["react", "typescript", "oxc"],
  "options": {
    "typeAware": true
  },
  "rules": {
    "react/rules-of-hooks": "error",
    "react/only-export-components": ["warn", { "allowConstantExport": true }]
  }
}
```

See the [Oxlint rules documentation](https://oxc.rs/docs/guide/usage/linter/rules) for the full list of rules and categories.
